    46→interface RouteInfo {
    47→  id: string;
    48→  name: string;
    49→  durationMinutes: number;

    53→
    54→interface BookingRow {
    55→  id: string;
    56→  code: string;
    57→  stageName: string | null;
    58→  status: string;
    59→  cashDue: number;
    60→  isCharter: boolean;

    64→
    65→function coastOf(routeName: string): "north" | "south" {
    66→  return routeName.toLowerCase().includes("south") ? "south" : "north";
    67→}
    68→
    69→export default function Home() {
    70→  const { me, loading: meLoading, refresh } = useMe();
    71→  const router = useRouter();
    72→  const splashSeen = useSplashOnce();
    73→  const [splashDone, setSplashDone] = useState(false);
    74→  const [loginTick, setLoginTick] = useState(0);
    75→
    76→  const showSplash = splashSeen && !splashDone;
    77→
    78→  const [direction, setDirection] = useState<"FROM_TERMINUS" | "TO_TERMINUS">("FROM_TERMINUS");
    79→  const [date, setDate] = useState(() => {
    80→    const d = new Date();
    81→    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    82→  });
    83→  const [routes, setRoutes] = useState<RouteInfo[]>([]);
    84→  const [trips, setTrips] = useState<BookableTrip[] | null>(null);
    85→  const [myBookings, setMyBookings] = useState<BookingRow[]>([]);
    86→  const [loading, setLoading] = useState(false);
    87→
    88→  // selection state (original store equivalents)
    89→  const [trainKey, setTrainKey] = useState<string | null>(null);
    90→  const [pointId, setPointId] = useState<string | null>(null);
    91→  const [homePickup, setHomePickup] = useState(false);
    92→  const [charter, setCharter] = useState(false);
    93→  const [seats, setSeats] = useState(1);
    94→
    95→  const [draft, setDraft] = useState<BookingDraft | null>(null);
    96→  const [sheetOpen, setSheetOpen] = useState(false);
    97→  const [payFor, setPayFor] = useState<BookingRow | null>(null);
    98→
    99→  const isAuthed = !!me?.session;
   100→
   101→  const loadTrips = useCallback(async (dir: string, d: string) => {
   102→    setLoading(true);
   103→    try {
   104→      const res = await api<{ trips: BookableTrip[] }>(`/api/trips?date=${d}&direction=${dir}`);
   105→      setTrips(res.trips);
   106→    } catch {
   107→      setTrips([]);
   108→    } finally {
   109→      setLoading(false);
   110→    }
   111→  }, []);
   112→
   113→  const loadMyBookings = useCallback(async () => {
   114→    try {
   115→      const res = await api<{ bookings: BookingRow[] }>("/api/bookings");
   116→      setMyBookings(res.bookings.filter((b) => ["awaiting_payment", "confirmed", "boarded"].includes(b.status)));
   117→    } catch {
   118→      setMyBookings([]);
   119→    }
   120→  }, []);
   121→
   122→  useEffect(() => {
   123→    api<{ routes: RouteInfo[] }>("/api/routes")
   124→      .then((r) => setRoutes(r.routes))
   125→      .catch(() => {});
   126→  }, []);
   127→
   128→  useEffect(() => {
   129→    if (isAuthed) loadMyBookings();
   130→  }, [isAuthed, loadMyBookings]);
   131→
   132→  useEffect(() => {
   133→    if (showSplash || !isAuthed) return;
   134→    loadTrips(direction, date);
   135→  }, [showSplash, isAuthed, direction, date, loadTrips]);
   136→
   137→  function openBooking(trip: BookableTrip) {
   138→    const fallbackStage =
   139→      direction === "FROM_TERMINUS"
   140→        ? trip.stages.filter((s) => s.order > 0).slice(-1)[0]
   141→        : trip.stages.filter((s) => s.order > 0)[0];
   142→    setDraft({
   143→      trip,
   144→      stageId: pointId && trip.stages.some((s) => s.id === pointId) ? pointId : fallbackStage?.id || "",
   145→      seats,
   146→      charter,
   147→      homePickup,
   148→      homeAddress: "",
   149→      useCredit: false,
   150→    });
   151→    setSheetOpen(true);
   152→  }
   153→
   154→  /* ─── Derived data (all hooks run before any early return) ──────────────── */
   155→
   156→  // Train pills from this direction's trips (unique MTM events)
   157→  const trainOptions = useMemo(() => {
   158→    const seen = new Map<string, { key: string; time: string; name: string; label: string }>();
   159→    for (const t of trips || []) {
   160→      if (!t.train) continue;
   161→      const key = `${t.train.name}-${t.train.mtmTime}`;
   162→      if (!seen.has(key)) {
   163→        seen.set(key, {
   164→          key,
   165→          time: t.train.mtmTime,
   166→          name: t.train.name,
   167→          label: t.direction === "FROM_TERMINUS" ? "Arrives MTM" : "Departs MTM",
   168→        });
   169→      }
   170→    }
   171→    return Array.from(seen.values());
   172→  }, [trips]);
   173→
   174→  const pointLabelCap = direction === "FROM_TERMINUS" ? "Drop-off" : "Pickup";
   175→
   176→  // Points grid grouped by route (coast), from the stable routes network
   177→  const pointGroups = useMemo(() => {
   178→    return routes
   179→      .map((r) => ({
   180→        route: r,
   181→        coast: coastOf(r.name),
   182→        points: r.stages.filter((s) => s.order > 0),
   183→      }))
   184→      .filter((g) => g.points.length > 0);
   185→  }, [routes]);
   186→
   187→  const selectedPoint = useMemo(() => {
   188→    for (const g of pointGroups) {
   189→      const p = g.points.find((s) => s.id === pointId);
   190→      if (p) return { point: p, route: g.route, coast: g.coast };
   191→    }
   192→    return null;
   193→  }, [pointGroups, pointId]);
   194→
   195→  // Trips after train + point filters
   196→  const visibleTrips = useMemo(() => {
   197→    let list = trips || [];
   198→    if (trainKey) list = list.filter((t) => t.train && `${t.train.name}-${t.train.mtmTime}` === trainKey);
   199→    if (pointId) list = list.filter((t) => t.stages.some((s) => s.id === pointId));
   200→    return list;
   201→  }, [trips, trainKey, pointId]);
   202→
   203→  // Live fare preview
   204→  const previewStage = selectedPoint?.point;
   205→  const farePreview = useMemo(() => {
   206→    const base = charter
   207→      ? selectedPoint?.route.charterPrice ?? 0
   208→      : (previewStage?.fare || 0) * seats;

   210→      ? (charter ? previewStage?.homeSurcharge || 0 : (previewStage?.homeSurcharge || 0) * seats)
   211→      : 0;
   212→    return { base, surcharge, total: base + surcharge };
   213→  }, [charter, previewStage, selectedPoint, seats, homePickup]);
   214→
   215→  /* ─── Splash & login gate (after all hooks) ──────────────────────────────── */
   216→
   217→  if (showSplash) {
   218→    return <SplashScreen onDone={() => setSplashDone(true)} />;
   219→  }
   220→
   221→  if (!meLoading && !isAuthed) {
   222→    return <LoginScreen key={loginTick} onDone={() => { setLoginTick((t) => t + 1); refresh(); }} />;
   223→  }
   224→
   225→  return (
   226→    <div className="flex min-h-screen flex-col bg-background">
   227→      <SiteHeader />
   228→
   229→      {/* Direction switcher — the signature pill row */}
   230→      <div className="mx-auto max-w-md w-full px-4 pt-3">
   231→        <Tabs value={direction} onValueChange={(v) => setDirection(v as "FROM_TERMINUS" | "TO_TERMINUS")}>
   232→          <TabsList className="grid w-full grid-cols-2">
   233→            <TabsTrigger value="FROM_TERMINUS" className="flex items-center gap-1.5">
   234→              <Anchor className="h-3.5 w-3.5" /> From Terminus
   235→            </TabsTrigger>
   236→            <TabsTrigger value="TO_TERMINUS" className="flex items-center gap-1.5">
   237→              <Navigation className="h-3.5 w-3.5" /> To Terminus
   238→            </TabsTrigger>
   239→          </TabsList>
   240→        </Tabs>
   241→      </div>
   242→      <SiteNav />
   243→
   244→      <main className="mx-auto w-full max-w-md flex-1 px-4 pb-6">
   245→        <AnimatePresence mode="wait">
   246→          <motion.div
   247→            key={direction}
   248→            initial={{ opacity: 0, y: 8 }}
   249→            animate={{ opacity: 1, y: 0 }}
   250→            exit={{ opacity: 0, y: -8 }}
   251→            transition={{ duration: 0.18 }}
   252→            className="space-y-4"
   253→          >
   254→            {/* Date */}
   255→            <div>
   256→              <div className="mb-2 flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   257→                <CalendarDays className="h-3.5 w-3.5" />
   258→                {direction === "FROM_TERMINUS" ? "Travelling on which date?" : "Catching the train on which date?"}
   259→              </div>
   260→              <DatePicker value={date} onChange={setDate} />
   261→              <div className="mt-1.5 text-xs text-muted-foreground">
   262→                Selected: <span className="font-medium text-foreground">{fmtDateShort(date)}</span>
   263→              </div>
   264→            </div>
   265→
   266→            {/* Train selector */}
   267→            <div>
   268→              <div className="mb-2 flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   269→                <TrainIcon className="h-3.5 w-3.5" />
   270→                {direction === "FROM_TERMINUS" ? "Arriving on which train?" : "Catching which train?"}
   271→              </div>
   272→              {trainOptions.length === 0 && !loading && (
   273→                <p className="rounded-lg border border-dashed p-3 text-xs text-muted-foreground">
   274→                  No train-linked departures for this day — try another date.
   275→                </p>

   277→              <div className="-mx-1 flex gap-2 overflow-x-auto px-1 pb-1">
   278→                {trainOptions.map((t) => (
   279→                  <TrainPill
   280→                    key={t.key}
   281→                    time={t.time}
   282→                    label={t.label}

   284→                    onClick={() => setTrainKey(trainKey === t.key ? null : t.key)}
   285→                  />
   286→                ))}
   287→              </div>
   288→            </div>
   289→
   290→            {/* Point picker */}
   291→            <div>
   292→              <div className="mb-2 flex items-center justify-between">
   293→                <div className="flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   294→                  <MapPin className="h-3.5 w-3.5" /> {pointLabelCap} point
   295→                </div>
   296→                {pointId && (
   297→                  <button onClick={() => setPointId(null)} className="text-[11px] text-primary hover:underline">
   298→                    Show all points
   299→                  </button>
   300→                )}
   301→              </div>
   302→              <div className="space-y-2">
   303→                {pointGroups.length === 0 && (
   304→                  <Skeleton className="h-20 w-full" />
   305→                )}
   306→                {pointGroups.map(({ route, coast, points }) => (
   307→                  <div key={route.id}>
   308→                    <div className="mb-1 flex items-center gap-1.5 text-[11px] font-medium text-muted-foreground">
   309→                      {coast === "south" ? (
   310→                        <span className="inline-flex rounded bg-orange-100 px-1.5 py-0.5 text-[9px] uppercase text-orange-800">
   311→                          South Coast
   312→                        </span>
   313→                      ) : (
   314→                        <span className="inline-flex rounded bg-emerald-100 px-1.5 py-0.5 text-[9px] uppercase text-emerald-800">
   315→                          North Coast
   316→                        </span>
   317→                      )}
   318→                      <span>{route.name.replace("Mombasa Terminus ↔ ", "").replace(" (Mtwapa · Malindi)", "")}</span>
   319→                    </div>
   320→                    <div className="grid grid-cols-2 gap-1.5">
   321→                      {points.map((p) => {
   322→                        const active = pointId === p.id;
   323→                        return (
   324→                          <button
   325→                            key={p.id}
   326→                            onClick={() => setPointId(active ? null : p.id)}
   327→                            className={cn(
   328→                              "rounded-lg border p-2 text-left text-xs transition-all",
   329→                              active
   330→                                ? "border-primary bg-primary shadow-sm text-primary-foreground"
   331→                                : "border-border bg-card hover:bg-accent hover:text-accent-foreground",
   332→                            )}
   333→                          >
   334→                            <div className="truncate font-medium">{p.name}</div>
   335→                            <div className={cn("mt-0.5 text-[10px]", active ? "opacity-80" : "text-muted-foreground")}>
   336→                              KSh {p.fare} · door +{p.homeSurcharge}
   337→                            </div>
   338→                          </button>
   339→                        );
   340→                      })}
   341→                    </div>
   342→                  </div>
   343→                ))}
   344→              </div>
   345→            </div>
   346→
   347→            {/* Options: door-to-door + charter + live fare preview */}
   348→            <Card className="border-primary/30 bg-primary/5">
   349→              <CardHeader className="pb-2">
   350→                <CardTitle className="flex items-center gap-1.5 text-sm">
   351→                  <Car className="h-4 w-4" /> {pointLabelCap} options
   352→                </CardTitle>
   353→              </CardHeader>
   354→              <CardContent className="space-y-3 pt-0">
   355→                {/* Door-to-door toggle */}
   356→                <div className="flex items-start justify-between gap-3 rounded-lg border bg-card p-3">
   357→                  <div className="flex-1">
   358→                    <Label className="flex items-center gap-1 text-sm font-medium">
   359→                      <HomeIcon className="h-4 w-4" /> Door-to-door {direction === "FROM_TERMINUS" ? "drop-off" : "pickup"}
   360→                    </Label>
   361→                    <p className="mt-1 text-[11px] text-muted-foreground">
   362→                      {selectedPoint
   363→                        ? `+${ksh(selectedPoint.point.homeSurcharge)}/seat — passes to your driver in full.`
   364→                        : "Pick a point first to see its door surcharge."}
   365→                    </p>
   366→                  </div>
   367→                  <Switch checked={homePickup} onCheckedChange={setHomePickup} aria-label="Door-to-door" />
   368→                </div>
   369→
   370→                {/* Charter toggle */}
   371→                <div className="flex items-start justify-between gap-3 rounded-lg border bg-card p-3">
   372→                  <div className="flex-1">
   373→                    <Label className="flex items-center gap-1 text-sm font-medium">
   374→                      <Crown className="h-4 w-4 text-violet-600" /> Book the whole vehicle (private)
   375→                    </Label>
   376→                    <p className="mt-1 text-[11px] text-muted-foreground">
   377→                      Reserve the entire cab for your family or group — no other passengers join.
   378→                    </p>
   379→                  </div>
   380→                  <Switch
   381→                    checked={charter}
   382→                    onCheckedChange={(c) => {
   383→                      setCharter(c);
   384→                      if (c) setHomePickup(false);
   385→                    }}
   386→                    aria-label="Charter"
   387→                  />
   388→                </div>
   389→
   390→                {/* Live fare preview */}
   391→                <div className="space-y-1.5 rounded-lg border bg-card p-3">
   392→                  <div className="text-[10px] uppercase tracking-wide text-muted-foreground">Live fare preview</div>
   393→                  <div className="space-y-1 text-xs">
   394→                    <Row
   395→                      label={charter ? "Charter (whole cab)" : `Base (per seat${selectedPoint ? ` · ${selectedPoint.point.name}` : ""})`}
   396→                      value={selectedPoint || charter ? ksh(charter ? (selectedPoint?.route.charterPrice ?? 0) : (previewStage?.fare || 0)) : "—"}
   397→                    />
   398→                    {homePickup && selectedPoint && (
   399→                      <Row label={`Door-to-door surcharge${charter ? "" : ` × ${seats}`}`} value={`+${ksh(previewStage?.homeSurcharge || 0)}${charter ? "" : ` × ${seats}`}`} />
   400→                    )}
   401→                    {!charter && seats > 1 && selectedPoint && <Row label={`Seats × ${seats}`} value={`${ksh(previewStage?.fare || 0)} × ${seats}`} />}
   402→                  </div>
   403→                  <div className="flex items-center justify-between border-t pt-1.5">
   404→                    <span className="text-sm font-medium">
   405→                      Total {charter ? "(whole cab)" : seats > 1 ? `(${seats} seats)` : ""}
   406→                    </span>
   407→                    <span className="text-lg font-bold tabular-nums text-primary">
   408→                      {selectedPoint || charter ? ksh(farePreview.total) : "—"}
   409→                    </span>
   410→                  </div>
   411→                </div>
   412→              </CardContent>
   413→            </Card>
   414→
   415→            {/* Seats (pooled only) */}
   416→            {!charter && (
   417→              <div>
   418→                <div className="mb-2 flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   419→                  <Users className="h-3.5 w-3.5" /> How many seats?
   420→                </div>
   421→                <SeatStepper value={seats} onChange={setSeats} min={1} max={14} />

   423→            )}
   424→
   425→            {/* Available cabs */}
   426→            <div className="space-y-2">
   427→              <div className="flex items-center justify-between">
   428→                <h3 className="text-sm font-medium">
   429→                  {charter ? "Available for charter" : "Available pooled cabs"}
   430→                </h3>
   431→                <span className="text-xs text-muted-foreground">{loading ? "…" : `${visibleTrips.length} cabs`}</span>
   432→              </div>
   433→
   434→              {loading && (
   435→                <div className="space-y-2">
   436→                  <Skeleton className="h-36 w-full" />
   437→                  <Skeleton className="h-36 w-full" />
   438→                </div>
   439→              )}
   440→
   441→              {!loading && visibleTrips.length === 0 && (
   442→                <Card>
   443→                  <CardContent className="py-8 text-center text-sm text-muted-foreground">
   444→                    {pointId
   445→                      ? "No cabs serve this point for that train. Try another point or \"Show all points\"."
   446→                      : "No cabs posted for this train yet."}
   447→                  </CardContent>
   448→                </Card>
   449→              )}
   450→
   451→              {!loading &&
   452→                visibleTrips.map((trip) => (
   453→                  <motion.div key={trip.id} layout whileTap={{ scale: 0.99 }}>
   454→                    <CabCard
   455→                      trip={trip}
   456→                      pointName={selectedPoint?.point.name}
   457→                      pointFare={selectedPoint?.point.fare}
   458→                      seats={seats}
   459→                      charter={charter}
   460→                      onBook={() => openBooking(trip)}
   461→                    />
   462→                  </motion.div>
   463→                ))}
   464→            </div>
   465→
   466→            {/* How it works */}
   467→            <Card className="border-dashed">
   468→              <CardHeader className="pb-2">
   469→                <CardTitle className="flex items-center gap-1.5 text-xs">
   470→                  <Info className="h-3.5 w-3.5" /> How points &amp; fares work
   471→                </CardTitle>
   472→              </CardHeader>
   473→              <CardContent className="space-y-1.5 pt-0 text-xs text-muted-foreground">
   474→                <p>
   475→                  Cabs meet every Madaraka Express at Mombasa Terminus. {direction === "FROM_TERMINUS" ? "Choose your drop-off point — the cab drops you there on the way from the terminus." : "Choose your pickup point and be there 10 minutes before the cab leaves for the terminus."}
   476→                </p>
   477→                <p>
   478→                  Shared-ride fare is <span className="font-medium text-foreground">KSh 400</span> per seat upcountry (Malindi 700, Diani Naivas 500). Door-to-door adds the point&apos;s surcharge, which goes to your driver in full.
   479→                </p>
   480→                <p>Private charter = the whole cab. Cancel before lock for a full refund; after lock, fares become 30-day travel credit.</p>
   481→              </CardContent>
   482→            </Card>
   483→
   484→            {/* My bookings */}
   485→            {myBookings.length > 0 && (
   486→              <div className="space-y-2 pt-2">
   487→                <h3 className="flex items-center gap-1.5 text-sm font-medium">
   488→                  <TicketIcon className="h-4 w-4" /> My bookings
   489→                </h3>
   490→                {myBookings.map((b) => (
   491→                  <Card key={b.id} className={cn(b.isCharter && "border-violet-300 bg-violet-50/30", b.status === "awaiting_payment" && "border-amber-300 bg-amber-50/30")}>
   492→                    <CardContent className="flex items-center justify-between gap-3 p-3">
   493→                      <div className="min-w-0">
   494→                        <div className="flex flex-wrap items-center gap-2 text-sm font-medium">
   495→                          <span>{b.stageName ?? b.code}</span>
   496→                          {b.isCharter && <CharterBadge />}
   497→                          <DotBadge status={b.status} />
   498→                        </div>
   499→                        <div className="mt-0.5 text-xs text-muted-foreground">
   500→                          {b.departureAt ? fmtTime(b.departureAt) : "awaiting allocation"}
   501→                          {b.driver ? ` · ${b.driver.name} · ${b.driver.plate}` : ""}
   502→                        </div>
   503→                        <div className="mt-0.5 text-xs font-medium">
   504→                          {ksh(b.cashDue)}
   505→                          {b.status === "awaiting_payment" && <span className="font-normal text-muted-foreground"> by M-Pesa</span>}
   506→                        </div>
   507→                      </div>
   508→                      {b.status === "awaiting_payment" ? (
   509→                        <Button size="sm" className="h-8 shrink-0" onClick={() => setPayFor(b)}>
   510→                          Pay now
   511→                        </Button>
   512→                      ) : (
   513→                        <Button size="sm" variant="outline" className="h-8 shrink-0" onClick={() => router.push("/bookings")}>
   514→                          Manage
   515→                        </Button>
   516→                      )}
   517→                    </CardContent>
   518→                  </Card>
   519→                ))}
   520→              </div>
   521→            )}
   522→          </motion.div>
   523→        </AnimatePresence>
   524→      </main>
   525→
   526→      <SiteFooter />
   527→
   528→      <BookingSheet
   529→        open={sheetOpen}
   530→        onClose={() => {
   531→          setSheetOpen(false);
   532→          loadTrips(direction, date);
   533→          loadMyBookings();
   534→        }}
   535→        draft={draft}
   536→        onBooked={() => {
   537→          loadTrips(direction, date);
   538→          loadMyBookings();
   539→        }}
   540→      />
   541→
   542→      {payFor && (
   543→        <PaySheet
   544→          open={!!payFor}
   545→          onClose={() => {
   546→            setPayFor(null);
   547→            loadMyBookings();
   548→          }}
   549→          booking={{
   550→            id: payFor.id,
   551→            code: payFor.code,
   552→            cashDue: payFor.cashDue,
   553→            stageName: payFor.stageName,
   554→            departureAt: payFor.departureAt,
   555→          }}
   556→          onPaid={loadMyBookings}
   557→        />
   558→      )}
   559→    </div>
   560→  );
   561→}

   572→function CabCard({
   573→  trip,
   574→  pointName,
   575→  pointFare,
   576→  seats,
   577→  charter,
   578→  onBook,
   579→}: {
   580→  trip: BookableTrip;
   581→  pointName?: string;
   582→  pointFare?: number;
   583→  seats: number;
   584→  charter: boolean;
   585→  onBook: () => void;
   586→}) {
   587→  const booked = trip.capacity - trip.seatsLeft;
   588→  const fare = charter ? trip.charterPrice : (pointFare ?? trip.minFare) * seats;
   589→  const full = trip.seatsLeft <= 0 || !trip.bookable;
   590→
   591→  return (
   592→    <Card className="overflow-hidden">
   593→      <CardContent className="space-y-2.5 p-3">
   594→        <div className="flex items-start justify-between gap-2">
   595→          <div className="min-w-0 flex-1">
   596→            <div className="flex flex-wrap items-center gap-2">
   597→              <span className="text-sm font-semibold">{pointName ?? trip.routeName}</span>
   598→              <DotBadge status={trip.status} />
   599→              {trip.status === "locked" && (
   600→                <span className="inline-flex items-center gap-0.5 rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] text-amber-900">
   601→                  <Sparkles className="h-2.5 w-2.5" /> credits only
   602→                </span>
   603→              )}
   604→            </div>
   605→            {trip.driver && (
   606→              <div className="mt-0.5 flex items-center gap-2 text-xs text-muted-foreground">
   607→                <span>{trip.driver.name}</span>
   608→                <Stars rating={trip.driver.rating} />
   609→                <span className="text-muted-foreground/50">•</span>
   610→                <span className="font-mono">{trip.driver.plate}</span>
   611→              </div>
   612→            )}
   613→            <div className="mt-0.5 flex items-center gap-1 text-[11px] text-muted-foreground">
   614→              <Clock className="h-3 w-3" /> Cab departs {fmtTime(trip.departureAt)}
   615→            </div>
   616→          </div>
   617→          <div className="shrink-0 text-right">
   618→            <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   619→              {charter ? "Charter" : seats > 1 ? `${seats} seats` : "Fare"}
   620→            </div>
   621→            <div className="font-bold tabular-nums text-primary">{ksh(fare)}</div>
   622→            {!charter && seats === 1 && pointFare != null && pointFare !== trip.minFare && (
   623→              <div className="text-[10px] text-muted-foreground">from {ksh(trip.minFare)}</div>
   624→            )}
   625→          </div>
   626→        </div>
   627→
   628→        <div className="flex items-center justify-between gap-3">
   629→          <div className="flex min-w-0 flex-wrap items-center gap-1.5 text-xs">
   630→            {trip.train && (
   631→              <span className="inline-flex items-center gap-1 rounded-full bg-secondary px-2 py-0.5 text-[11px] font-medium text-secondary-foreground">
   632→                <TrainIcon className="h-3 w-3" />
   633→                {trip.direction === "FROM_TERMINUS"
   634→                  ? `Meets the ${trip.train.name} · MTM ${trip.train.mtmTime}`
   635→                  : `Catches the ${trip.train.name} · MTM ${trip.train.mtmTime}`}
   636→              </span>
   637→            )}
   638→            {trip.direction === "TO_TERMINUS" && (
   639→              <span className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   640→                <Navigation className="h-3 w-3" /> reaches MTM {fmtTime(trip.terminusAt)}
   641→              </span>
   642→            )}
   643→          </div>
   644→          <SeatMeter booked={booked} capacity={trip.capacity} />
   645→        </div>
   646→
   647→        <Button className="h-9 w-full" size="sm" disabled={full} onClick={onBook}>
   648→          {full
   649→            ? trip.seatsLeft <= 0
   650→              ? "Not enough seats"
   651→              : "Unavailable"
   652→            : charter
   653→              ? `Book charter · ${ksh(trip.charterPrice)}`
   654→              : `Reserve ${seats > 1 ? `${seats} seats` : "seat"} · ${ksh(fare)}`}
   655→        </Button>
   656→      </CardContent>
   657→    </Card>
   658→  );
   659→}
