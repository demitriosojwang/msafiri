   572→   572→function CabCard({
   573→   573→  trip,
   574→   574→  pointName,
   575→   575→  pointFare,
   576→   576→  seats,
   577→   577→  charter,
   578→   578→  onBook,
   579→   579→}: {
   580→   580→  trip: BookableTrip;
   581→   581→  pointName?: string;
   582→   582→  pointFare?: number;
   583→   583→  seats: number;
   584→   584→  charter: boolean;
   585→   585→  onBook: () => void;
   586→   586→}) {
   587→   587→  const booked = trip.capacity - trip.seatsLeft;
   588→   588→  const fare = charter ? trip.charterPrice : (pointFare ?? trip.minFare) * seats;
   589→   589→  const full = trip.seatsLeft <= 0 || !trip.bookable;
   590→   590→
   591→   591→  return (
   592→   592→    <Card className="overflow-hidden">
   593→   593→      <CardContent className="space-y-2.5 p-3">
   594→   594→        <div className="flex items-start justify-between gap-2">
   595→   595→          <div className="min-w-0 flex-1">
   596→   596→            <div className="flex flex-wrap items-center gap-2">
   597→   597→              <span className="text-sm font-semibold">{pointName ?? trip.routeName}</span>
   598→   598→              <DotBadge status={trip.status} />
   599→   599→              {trip.status === "locked" && (
   600→   600→                <span className="inline-flex items-center gap-0.5 rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] text-amber-900">
   601→   601→                  <Sparkles className="h-2.5 w-2.5" /> credits only
   602→   602→                </span>
   603→   603→              )}
   604→   604→            </div>
   605→   605→            {trip.driver && (
   606→   606→              <div className="mt-0.5 flex items-center gap-2 text-xs text-muted-foreground">
   607→   607→                <span>{trip.driver.name}</span>
   608→   608→                <Stars rating={trip.driver.rating} />
   609→   609→                <span className="text-muted-foreground/50">•</span>
   610→   610→                <span className="font-mono">{trip.driver.plate}</span>
   611→   611→              </div>
   612→   612→            )}
   613→   613→            <div className="mt-0.5 flex items-center gap-1 text-[11px] text-muted-foreground">
   614→   614→              <Clock className="h-3 w-3" /> Cab departs {fmtTime(trip.departureAt)}
   615→   615→            </div>
   616→   616→          </div>
   617→   617→          <div className="shrink-0 text-right">
   618→   618→            <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   619→   619→              {charter ? "Charter" : seats > 1 ? `${seats} seats` : "Fare"}
   620→   620→            </div>
   621→   621→            <div className="font-bold tabular-nums text-primary">{ksh(fare)}</div>
   622→   622→            {!charter && seats === 1 && pointFare != null && pointFare !== trip.minFare && (
   623→   623→              <div className="text-[10px] text-muted-foreground">from {ksh(trip.minFare)}</div>
   624→   624→            )}
   625→   625→          </div>
   626→   626→        </div>
   627→   627→
   628→   628→        <div className="flex items-center justify-between gap-3">
   629→   629→          <div className="flex min-w-0 flex-wrap items-center gap-1.5 text-xs">
   630→   630→            {trip.train && (
   631→   631→              <span className="inline-flex items-center gap-1 rounded-full bg-secondary px-2 py-0.5 text-[11px] font-medium text-secondary-foreground">
   632→   632→                <TrainIcon className="h-3 w-3" />

   634→   634→                  ? `Meets the ${trip.train.name} · MTM ${trip.train.mtmTime}`

   652→   652→            : charter
