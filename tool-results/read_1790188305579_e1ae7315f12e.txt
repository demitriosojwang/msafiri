     4→/**
     5→ * M-Pesa Daraja client — dual mode.
     6→ *
     7→ * RESOLUTION ORDER for credentials: environment variables first, then the
     8→ * credentials saved in Admin → Payments (PlatformConfig). When the four core
     9→ * fields exist (consumer key, consumer secret, shortcode, passkey) the client
    10→ * goes LIVE against Safaricom Daraja; otherwise every call falls through to
    11→ * the deterministic demo simulator, so the whole platform stays demonstrable
    12→ * before real API access arrives.
    13→ *
    14→ * Applying real credentials is therefore zero-code: paste them in the admin
    15→ * console (or set the env vars) and press "Test connection". Nothing else
    16→ * changes — result-code contracts are identical in both modes.
    17→ *
    18→ * Capabilities (same contract in both modes):
    19→ *   - stkPush            → fare collection (Lipa na M-Pesa Online / STK push)
    20→ *   - queryStkResult     → poll the outcome of a push (verify step)
    21→ *   - transactionStatus  → missed-callback recovery sweep
    22→ *   - reversal           → full refunds inside the short window
    23→ *   - b2c                → policy refunds outside the window + driver payouts
    24→ *
    25→ * Environment variables (all optional, override the saved credentials):
    26→ *   MPESA_MODE                force "mock" even when credentials exist
    27→ *   MPESA_ENVIRONMENT         sandbox | production
    28→ *   MPESA_CONSUMER_KEY        Daraja app consumer key
    29→ *   MPESA_CONSUMER_SECRET     Daraja app consumer secret
    30→ *   MPESA_SHORTCODE           collection (head) shortcode — fares land here
    31→ *   MPESA_PASSKEY             Lipa na M-Pesa Online passkey
    32→ *   MPESA_CALLBACK_BASE_URL   public https base, e.g. https://mireli.co.ke
    33→ *   MPESA_B2C_SHORTCODE       initiator shortcode for B2C/reversal (optional)
    34→ *   MPESA_INITIATOR_NAME      API operator username for B2C/reversal
    35→ *   MPESA_INITIATOR_PASSWORD  encrypted with the Safaricom cert at call time
    36→ *   MPESA_SECURITY_CREDENTIAL OR a pre-encrypted security credential
    37→ *   MPESA_CERT                Safaricom public certificate (PEM)
    38→ *
    39→ * Demo-simulator extras (mock mode only):
    40→ *   - Any receiver whose number is in INVALID_B2C_NUMBERS fails B2C with code
    41→ *     "1001" — demonstrates payout-failure handling (flagged, never retried).
    42→ *   - The admin Reconciliation page can inject one refund/payout failure and
    43→ *     a "missed callback" the nightly Transaction Status sweep recovers.
    44→ */
    45→

    67→export function generateMpesaReceipt(): string {
    68→  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ0123456789";
    69→  let s = "";
    70→  for (let i = 0; i < 10; i++) s += chars[crypto.randomInt(chars.length)];
    71→  return s;

    77→  return `MR-${s}`;

   138→/** Resolve credentials: env vars override the admin-saved ones. */
   139→export async function resolveDaraja(): Promise<DarajaCredentials> {
   140→  const cfg = await db.platformConfig.findUnique({ where: { id: "main" } });
   141→  const env = envCredentials();
   142→
   143→  const pick = (envVal: string | undefined, dbVal: string | undefined | null) =>
   144→    envVal !== undefined && envVal !== "" ? envVal : dbVal || "";
   145→
   146→  const creds: DarajaCredentials = {
   147→    source: env.consumerKey ? "environment" : cfg?.mpesaConsumerKey ? "database" : "none",
   148→    mode: "mock", // final value decided below
   149→    environment: (pick(env.environment, cfg?.mpesaEnvironment) || "sandbox") as string,
   150→    consumerKey: pick(env.consumerKey, cfg?.mpesaConsumerKey),
   151→    consumerSecret: pick(env.consumerSecret, cfg?.mpesaConsumerSecret),
   152→    shortcode: pick(env.shortcode, cfg?.mpesaShortcode),
   153→    passkey: pick(env.passkey, cfg?.mpesaPasskey),
   154→    callbackBaseUrl: pick(env.callbackBaseUrl, cfg?.mpesaCallbackBaseUrl),
   155→    b2cShortcode: pick(env.b2cShortcode, cfg?.mpesaB2CShortcode),
   156→    initiatorName: pick(env.initiatorName, cfg?.mpesaInitiatorName),
   157→    initiatorPassword: pick(env.initiatorPassword, cfg?.mpesaInitiatorPassword),
   158→    securityCredential: pick(env.securityCredential, cfg?.mpesaSecurityCredential),
   159→    cert: pick(env.cert, cfg?.mpesaCert),
   160→  };
   161→
   162→  const forcedMock = (process.env.MPESA_MODE || "").trim().toLowerCase() === "mock";
   163→  const complete =
   164→    !!creds.consumerKey && !!creds.consumerSecret && !!creds.shortcode && !!creds.passkey;
   165→  creds.mode = !forcedMock && complete ? "live" : "mock";
   166→  return creds;
   167→}

   199→/** 07XX / 2547XX / +2547XX → 2547XXXXXXXX (Daraja format), null if invalid. */
   200→export function toMpesaMsISDN(input: string): string | null {
   201→  const digits = input.replace(/\D/g, "");
   202→  if (/^254(7|1)\d{8}$/.test(digits)) return digits;
   203→  if (/^0(7|1)\d{8}$/.test(digits)) return `254${digits.slice(1)}`;
   204→  if (/^(7|1)\d{8}$/.test(digits)) return `254${digits}`;
   205→  return null;
   206→}

   208→/** Daraja timestamps are Nairobi time (EAT, UTC+3): YYYYMMDDHHmmss */
   209→function mpesaTimestamp(): string {
   210→  const eat = new Date(Date.now() + 3 * 60 * 60 * 1000);
   211→  const p = (n: number) => String(n).padStart(2, "0");
   212→  return (
   213→    `${eat.getUTCFullYear()}${p(eat.getUTCMonth() + 1)}${p(eat.getUTCDate())}` +
   214→    `${p(eat.getUTCHours())}${p(eat.getUTCMinutes())}${p(eat.getUTCSeconds())}`
   215→  );
   216→}

   218→function stkPassword(creds: DarajaCredentials, timestamp: string): string {
   219→  return Buffer.from(`${creds.shortcode}${creds.passkey}${timestamp}`).toString("base64");
   220→}
   221→
   222→/** Security credential for B2C/Reversal: pre-encrypted, or encrypt the
   223→ *  initiator password with the Safaricom public cert (RSA PKCS#1 v1.5). */
   224→function securityCredential(creds: DarajaCredentials): string | null {
   225→  if (creds.securityCredential) return creds.securityCredential;
   226→  if (creds.initiatorPassword && creds.cert) {
   227→    try {
   228→      const encrypted = crypto.publicEncrypt(
   229→        { key: certsToPemKey(creds.cert), padding: crypto.constants.RSA_PKCS1_PADDING },
   230→        Buffer.from(creds.initiatorPassword, "utf8"),
   231→      );
   232→      return encrypted.toString("base64");
   233→    } catch {
   234→      return null;
   235→    }
   236→  }
   237→  return null;
   238→}
   239→
   240→/** Safaricom publishes a X.509 CERT; crypto wants a public key — extract it. */
   241→function certsToPemKey(certPem: string): string {
   242→  if (certPem.includes("BEGIN PUBLIC KEY")) return certPem;
   243→  try {
   244→    const x509 = new crypto.X509Certificate(certPem);
   245→    return x509.publicKey.export({ type: "spki", format: "pem" }).toString();
   246→  } catch {
   247→    return certPem; // let publicEncrypt surface the real error
   248→  }
   249→}

   253→interface TokenCache {

   255→  token: string;
   256→  expiresAt: number;

   258→const gToken = globalThis as unknown as { __darajaToken?: TokenCache };

   260→async function darajaFetch(url: string, init: RequestInit): Promise<Response> {
   261→  const ctrl = new AbortController();
   262→  const timer = setTimeout(() => ctrl.abort(), 20_000);
   263→  try {
   264→    return await fetch(url, { ...init, signal: ctrl.signal, cache: "no-store" });
   265→  } finally {
   266→    clearTimeout(timer);
   267→  }
   268→}

   270→/** OAuth access token — cached until ~60s before expiry. Throws on failure. */
   271→export async function getAccessToken(creds: DarajaCredentials): Promise<string> {
   272→  const cacheKey = `${creds.environment}:${creds.consumerKey}`;
   273→  const cached = gToken.__darajaToken;
   274→  if (cached && cached.key === cacheKey && cached.expiresAt > Date.now()) return cached.token;
   275→
   276→  const auth = Buffer.from(`${creds.consumerKey}:${creds.consumerSecret}`).toString("base64");
   277→  const res = await darajaFetch(`${apiBase(creds)}/oauth/v1/generate?grant_type=client_credentials`, {
   278→    headers: { Authorization: `Basic ${auth}` },
   279→  });
   280→  if (!res.ok) {
   281→    const text = await res.text().catch(() => "");
   282→    throw new Error(`Daraja OAuth failed (${res.status}): ${text.slice(0, 200)}`);
   283→  }
   284→  const data = (await res.json()) as { access_token: string; expires_in: string };
   285→  gToken.__darajaToken = {
   286→    key: cacheKey,
   287→    token: data.access_token,
   288→    expiresAt: Date.now() + (parseInt(data.expires_in || "3599", 10) - 60) * 1000,
   289→  };
   290→  return data.access_token;
   291→}

   295→export interface StkPushResult {
   296→  ok: boolean;
   297→  checkoutRequestId: string;
   298→  merchantRequestId: string;
   299→  responseCode: string;
   300→  responseDescription: string;

   303→/** Initiates the STK push. Mock: accepted instantly (the Pay Sheet's verify
   304→ *  step simulates the customer). Live: real Daraja request; the customer
   305→ *  answers on their phone and Daraja posts the result to our callback. */
   306→export async function stkPush(phone: string, amount: number, ref: string): Promise<StkPushResult> {
   307→  if (amount <= 0) throw new Error("STK push amount must be positive");
   308→  const creds = await resolveDaraja();
   309→
   310→  if (creds.mode === "mock") {
   311→    return {
   312→      ok: true,
   313→      checkoutRequestId: generateCheckoutRequestId(),
   314→      merchantRequestId: generateMerchantRequestId(),
   315→      responseCode: "0",
   316→      responseDescription: `Success. Request accepted for processing — ref ${ref}`,
   317→    };
   318→  }
   319→
   320→  const party = toMpesaMsISDN(phone);
   321→  if (!party) {
   322→    return {
   323→      ok: false,
   324→      checkoutRequestId: "",
   325→      merchantRequestId: "",
   326→      responseCode: "400",
   327→      responseDescription: "Invalid M-Pesa number — use 07XX or 2547XX format",
   328→    };
   329→  }
   330→  const urls = callbackUrls(creds);
   331→  if (!urls.base) {
   332→    return {
   333→      ok: false,
   334→      checkoutRequestId: "",
   335→      merchantRequestId: "",
   336→      responseCode: "400",
   337→      responseDescription: "Callback base URL is not configured — set it in Admin → Payments",
   338→    };
   339→  }
   340→  const timestamp = mpesaTimestamp();
   341→  try {
   342→    const token = await getAccessToken(creds);
   343→    const res = await darajaFetch(`${apiBase(creds)}/mpesa/stkpush/v1/processrequest`, {
   344→      method: "POST",
   345→      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   346→      body: JSON.stringify({
   347→        BusinessShortCode: creds.shortcode,
   348→        Password: stkPassword(creds, timestamp),
   349→        Timestamp: timestamp,
   350→        TransactionType: "CustomerPayBillOnline",
   351→        Amount: Math.round(amount),
   352→        PartyA: party,
   353→        PartyB: creds.shortcode,
   354→        PhoneNumber: party,
   355→        CallBackURL: urls.stkResult,
   356→        AccountReference: ref.slice(0, 12),
   357→        TransactionDesc: `Mi-Reli fare ${ref}`.slice(0, 20),
   358→      }),
   359→    });
   360→    const data = (await res.json().catch(() => ({}))) as {
   361→      ResponseCode?: string;
   362→      ResponseDescription?: string;
   363→      errorMessage?: string;
   364→      CheckoutRequestID?: string;
   365→      MerchantRequestID?: string;
   366→    };
   367→    if (data.ResponseCode === "0" && data.CheckoutRequestID) {
   368→      return {
   369→        ok: true,
   370→        checkoutRequestId: data.CheckoutRequestID,
   371→        merchantRequestId: data.MerchantRequestID || "",
   372→        responseCode: "0",
   373→        responseDescription: data.ResponseDescription || "Success. Request accepted for processing",
   374→      };
   375→    }
   376→    return {
   377→      ok: false,
   378→      checkoutRequestId: "",
   379→      merchantRequestId: "",
   380→      responseCode: data.ResponseCode || String(res.status),
   381→      responseDescription: data.errorMessage || data.ResponseDescription || "STK push rejected by Daraja",
   382→    };
   383→  } catch (e) {
   384→    return {
   385→      ok: false,
   386→      checkoutRequestId: "",
   387→      merchantRequestId: "",
   388→      responseCode: RESULT_CODES.TIMEOUT,
   389→      responseDescription: `Could not reach Daraja: ${e instanceof Error ? e.message : "network error"}`,
   390→    };
   391→  }
   392→}

   403→function extractReceipt(items: { Name: string; Value?: unknown }[] | undefined): string | undefined {
   404→  const hit = items?.find((i) => i.Name === "MpesaReceiptNumber");
   405→  return typeof hit?.Value === "string" ? hit.Value : undefined;

   408→/** Poll the outcome of an STK push.
   409→ *  Mock: always succeeds (the demo "verify" tap = the customer's PIN).
   410→ *  Live: real stkpushquery — pending while the customer hasn't acted. */
   411→export async function queryStkResult(checkoutRequestId: string): Promise<StkQueryResult> {
   412→  const creds = await resolveDaraja();
   413→  if (creds.mode === "mock") {
   414→    return {
   415→      resultCode: RESULT_CODES.SUCCESS,
   416→      resultDesc: "The service request is processed successfully.",
   417→      mpesaReceipt: generateMpesaReceipt(),
   418→    };
   419→  }
   420→  const timestamp = mpesaTimestamp();
   421→  try {
   422→    const token = await getAccessToken(creds);
   423→    const res = await darajaFetch(`${apiBase(creds)}/mpesa/stkpushquery/v1/query`, {
   424→      method: "POST",
   425→      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   426→      body: JSON.stringify({
   427→        BusinessShortCode: creds.shortcode,
   428→        Password: stkPassword(creds, timestamp),
   429→        Timestamp: timestamp,
   430→        CheckoutRequestID: checkoutRequestId,
   431→      }),
   432→    });
   433→    const data = (await res.json().catch(() => ({}))) as {
   434→      ResultCode?: string;
   435→      ResultDesc?: string;
   436→      errorMessage?: string;
   437→      errorCode?: string;
   438→      CallbackMetadata?: { Item?: { Name: string; Value?: unknown }[] };
   439→    };
   440→    if (typeof data.ResultCode === "string") {
   441→      return {
   442→        resultCode: data.ResultCode,
   443→        resultDesc: data.ResultDesc || "Query result",
   444→        mpesaReceipt:
   445→          data.ResultCode === RESULT_CODES.SUCCESS
   446→            ? extractReceipt(data.CallbackMetadata?.Item)
   447→            : undefined,
   448→      };
   449→    }
   450→    // No terminal result yet — the customer hasn't answered (or ID unknown yet)
   451→    return {
   452→      resultCode: "",
   453→      resultDesc: data.errorMessage || data.errorCode || "Still processing — try again shortly",
   454→      pending: true,
   455→    };
   456→  } catch (e) {
   457→    return {
   458→      resultCode: "",
   459→      resultDesc: `Could not reach Daraja: ${e instanceof Error ? e.message : "network error"}`,
   460→      pending: true,
   461→    };
   462→  }
   463→}

   465→// ─── Transaction Status (missed-callback recovery) ───────────────────────────
   466→
   467→/** Resolve an ambiguous payment. ws_CO_* ids are polled via stkpushquery;
   468→ *  M-Pesa receipts go through the Transaction Status endpoint.
   469→ *  Mock: always resolves as paid — the demo point is that the sweep recovers
   470→ *  what a lost webhook would have dropped. */
   471→export async function transactionStatus(
   472→  transactionId: string,
   473→): Promise<StkQueryResult> {
   474→  const creds = await resolveDaraja();
   475→  if (creds.mode === "mock") {
   476→    return {
   477→      resultCode: RESULT_CODES.SUCCESS,
   478→      resultDesc: `Transaction completed successfully (recovered via status query for ${transactionId})`,
   479→      mpesaReceipt: generateMpesaReceipt(),
   480→    };
   481→  }
   482→  if (transactionId.startsWith("ws_CO_")) return queryStkResult(transactionId);
   483→
   484→  const initiator = securityCredential(creds);
   485→  if (!creds.initiatorName || !initiator) {
   486→    return {
   487→      resultCode: RESULT_CODES.INVALID_INITIATOR,
   488→      resultDesc: "Transaction Status needs the B2C initiator name + security credential (Admin → Payments)",
   489→    };
   490→  }
   491→  const urls = callbackUrls(creds);
   492→  try {
   493→    const token = await getAccessToken(creds);
   494→    const res = await darajaFetch(`${apiBase(creds)}/mpesa/transactionstatus/v1/query`, {
   495→      method: "POST",
   496→      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   497→      body: JSON.stringify({
   498→        Initiator: creds.initiatorName,
   499→        SecurityCredential: initiator,
   500→        CommandID: "TransactionStatusQuery",
   501→        TransactionID: transactionId,
   502→        PartyA: creds.b2cShortcode || creds.shortcode,
   503→        IdentifierType: "4",
   504→        ResultURL: urls.b2cResult,
   505→        QueueTimeOutURL: urls.stkTimeout,
   506→        Remarks: "Mi-Reli reconciliation sweep",
   507→        Occasion: "reconciliation",
   508→      }),
   509→    });
   510→    const data = (await res.json().catch(() => ({}))) as {
   511→      ResponseCode?: string;
   512→      ResponseDescription?: string;
   513→      errorMessage?: string;
   514→    };
   515→    if (data.ResponseCode === "0") {
   516→      // Accepted — the authoritative result lands on the b2c result webhook.
   517→      return {
   518→        resultCode: "",
   519→        resultDesc: "Status query accepted — result will arrive on the webhook",
   520→        pending: true,
   521→      };
   522→    }
   523→    return {
   524→      resultCode: data.ResponseCode || RESULT_CODES.TIMEOUT,
   525→      resultDesc: data.errorMessage || data.ResponseDescription || "Status query rejected",
   526→    };
   527→  } catch (e) {
   528→    return {
   529→      resultCode: "",
   530→      resultDesc: `Could not reach Daraja: ${e instanceof Error ? e.message : "network error"}`,
   531→      pending: true,
   532→    };
   533→  }
   534→}

   543→export async function reversal(params: {
   544→  transactionId: string; // original M-Pesa receipt
   545→  amount: number;
   546→}): Promise<DarajaResultCode> {
   547→  const creds = await resolveDaraja();
   548→  if (creds.mode === "mock") {
   549→    if (state.injectNextRefundFailure) {
   550→      state.injectNextRefundFailure = false;
   551→      return {
   552→        resultCode: RESULT_CODES.INVALID_INITIATOR,
   553→        resultDesc: "Reversal rejected: invalid initiator credentials (simulated failure)",
   554→      };
   555→    }
   556→    return {
   557→      resultCode: RESULT_CODES.SUCCESS,
   558→      resultDesc: `Reversal of KSh ${params.amount} on ${params.transactionId} accepted`,
   559→    };
   560→  }
   561→
   562→  const initiator = securityCredential(creds);
   563→  if (!creds.initiatorName || !initiator) {
   564→    return {
   565→      resultCode: RESULT_CODES.INVALID_INITIATOR,
   566→      resultDesc: "Reversal needs the B2C initiator name + security credential (Admin → Payments)",
   567→    };
   568→  }
   569→  const urls = callbackUrls(creds);
   570→  try {
   571→    const token = await getAccessToken(creds);
   572→    const res = await darajaFetch(`${apiBase(creds)}/mpesa/reversal/v1/request`, {
   573→      method: "POST",
   574→      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   575→      body: JSON.stringify({
   576→        Initiator: creds.initiatorName,
   577→        SecurityCredential: initiator,
   578→        CommandID: "TransactionReversal",
   579→        TransactionID: params.transactionId,
   580→        Amount: Math.round(params.amount),
   581→        ReceiverParty: creds.b2cShortcode || creds.shortcode,
   582→        RecieverIdentifierType: "11", // Safaricom's spelling — do not "fix"
   583→        ResultURL: urls.b2cResult,
   584→        QueueTimeOutURL: urls.stkTimeout,
   585→        Remarks: "Mi-Reli refund",
   586→      }),
   587→    });
   588→    const data = (await res.json().catch(() => ({}))) as {
   589→      ResponseCode?: string;
   590→      ResponseDescription?: string;
   591→      errorMessage?: string;
   592→    };
   593→    if (data.ResponseCode === "0") {
   594→      return {
   595→        resultCode: RESULT_CODES.SUCCESS,
   596→        resultDesc: data.ResponseDescription || "Reversal accepted",
   597→      };
   598→    }
   599→    return {
   600→      resultCode: data.ResponseCode || RESULT_CODES.TIMEOUT,
   601→      resultDesc: data.errorMessage || data.ResponseDescription || "Reversal rejected",
   602→    };
   603→  } catch (e) {
   604→    return {
   605→      resultCode: RESULT_CODES.TIMEOUT,
   606→      resultDesc: `Could not reach Daraja: ${e instanceof Error ? e.message : "network error"}`,
   607→    };
   608→  }
   609→}

   613→export async function b2c(params: {
   614→  receiverPhone: string;
   615→  amount: number;
   616→  remarks?: string;
   617→}): Promise<DarajaResultCode> {
   618→  const creds = await resolveDaraja();
   619→  if (creds.mode === "mock") {
   620→    if (state.injectNextPayoutFailure) {
   621→      state.injectNextPayoutFailure = false;
   622→      return {
   623→        resultCode: RESULT_CODES.INVALID_RECEIVER,
   624→        resultDesc: "B2C rejected: invalid receiver number (simulated failure)",
   625→      };
   626→    }
   627→    if (INVALID_B2C_NUMBERS.has(params.receiverPhone)) {
   628→      return {
   629→        resultCode: RESULT_CODES.INVALID_RECEIVER,
   630→        resultDesc: `B2C rejected: receiver ${params.receiverPhone} is invalid or deregistered`,
   631→      };
   632→    }
   633→    return {
   634→      resultCode: RESULT_CODES.SUCCESS,
   635→      resultDesc: `B2C of KSh ${params.amount} to ${params.receiverPhone} accepted`,
   636→    };
   637→  }
   638→
   639→  const party = toMpesaMsISDN(params.receiverPhone);
   640→  if (!party) {
   641→    return {
   642→      resultCode: RESULT_CODES.INVALID_RECEIVER,
   643→      resultDesc: `B2C rejected: receiver ${params.receiverPhone} is not a valid Safaricom number`,
   644→    };
   645→  }
   646→  const initiator = securityCredential(creds);
   647→  if (!creds.initiatorName || !initiator) {
   648→    return {
   649→      resultCode: RESULT_CODES.INVALID_INITIATOR,
   650→      resultDesc: "B2C needs the initiator name + security credential (Admin → Payments)",
   651→    };
   652→  }
   653→  const urls = callbackUrls(creds);
   654→  try {
   655→    const token = await getAccessToken(creds);
   656→    const res = await darajaFetch(`${apiBase(creds)}/mpesa/b2c/v3/paymentrequest`, {
   657→      method: "POST",
   658→      headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
   659→      body: JSON.stringify({
   660→        OriginatorConversationID: `MR-${Date.now()}-${crypto.randomInt(1000, 9999)}`,
   661→        InitiatorName: creds.initiatorName,
   662→        SecurityCredential: initiator,
   663→        CommandID: "BusinessPayment",
   664→        Amount: Math.round(params.amount),
   665→        PartyA: creds.b2cShortcode || creds.shortcode,
   666→        PartyB: party,
   667→        Remarks: (params.remarks || "Mi-Reli payout").slice(0, 100),
   668→        QueueTimeOutURL: urls.stkTimeout,
   669→        ResultURL: urls.b2cResult,
   670→      }),
   671→    });
   672→    const data = (await res.json().catch(() => ({}))) as {
   673→      ResponseCode?: string;
   674→      ResponseDescription?: string;
   675→      errorMessage?: string;
   676→    };
   677→    if (data.ResponseCode === "0") {
   678→      return {
   679→        resultCode: RESULT_CODES.SUCCESS,
   680→        resultDesc: data.ResponseDescription || "B2C accepted",
   681→      };
   682→    }
   683→    return {
   684→      resultCode: data.ResponseCode || RESULT_CODES.TIMEOUT,
   685→      resultDesc: data.errorMessage || data.ResponseDescription || "B2C rejected",
   686→    };
   687→  } catch (e) {
   688→    return {
   689→      resultCode: RESULT_CODES.TIMEOUT,
   690→      resultDesc: `Could not reach Daraja: ${e instanceof Error ? e.message : "network error"}`,
   691→    };
   692→  }
   693→}
